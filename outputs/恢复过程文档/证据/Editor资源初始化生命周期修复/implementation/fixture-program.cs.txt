class Waiter:IEnumerator,IDisposable {
 public int Moves,Disposals;public object Value;public bool End,EndAfterFirst,ThrowMove,ThrowDispose;

 public void Reset()=>throw new NotSupportedException();public void Dispose(){Disposals++;if(ThrowDispose)throw new IOException("dispose marker");}
}
